# og-ticket Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Claude Code mod that shows the project's work item in focus on the status line (Jira via `acli` or Atlassian Rovo, or a free-text `manual` focus) and posts a Claude-drafted comment to it only after the user confirms.

**Architecture:** Three files under `mods/og-ticket/hooks/`: `focus.ts` (pure: key from branch, focus resolution, status text), `sources.ts` (the `acli` and Rovo adapters behind one `Source` interface, with pure parsers), and `register.tsx` (events, `/og-ticket`, the `og_ticket_comment` tool, `$.store`). Same layout, idioms and test harness as `mods/og-sf-org-status`.

**Tech Stack:** Claude Code hooks API (`claude-code`, `claude-code/testing`), TypeScript, `claude plugin validate` / `claude plugin test`, `scripts/check.sh`.

**Spec:** `docs/specs/2026-10-14-og-ticket-design.md`

## Global Constraints

- Mod name `og-ticket`, folder `mods/og-ticket/`, command `/og-ticket`, tool `og_ticket_comment` (model sees `mcp__og-ticket__og_ticket_comment`).
- Sources: exactly `acli`, `rovo`, `manual`.
- Key regex: `[A-Z][A-Z0-9]+-\d+`.
- Status text: Jira `🎫 <KEY> · <Status> · <summary>`; unresolved `🎫 <KEY> · ?`; manual `🎯 <text>`; whole line cut to 60 chars with `…`.
- Item cache: 5 minutes. Branch poll: 60 s.
- A comment posts only after `$.ui.ask` answers `Post`, in every permission mode; with nobody to ask it refuses.
- Code, comments, README and commit messages in English (repo convention since `73be135`).
- Version `0.1.0`; add the mod to `.claude-plugin/marketplace.json`.

## Review Focus

1. Detached HEAD / not a git repo / `git` missing → no branch key, no error, line from the pin or cleared. (Task 3 test.)
2. `acli` not installed or not logged in (exit ≠ 0, ENOENT) → discovery falls through to Rovo, then `· ?`; nothing stored. (Task 3 test.)
3. Rovo connector not authenticated (no `getJiraIssue` tool listed) → treated as unavailable, never throws. (Task 2 test.)
4. Focus changes while a fetch is in flight (branch switch) → the stale result must not overwrite the new line. (Task 3 test.)
5. `og_ticket_comment` called with no focus, a `manual` focus, or an empty body → refused with a clear message, nothing posted. (Task 5 test.)

// Pure logic of the focus: which item, from the pin or the branch, and the
// status line for it. No `$` here, so tests drive it alone.

import type { Focus, Item, Pin } from '../types'

const KEY = /[A-Z][A-Z0-9]+-\d+/
const MAX = 60

/** The Jira key in a branch name (`feature/SFCOMPRG-1234-x` → `SFCOMPRG-1234`), if any. */
export const keyIn = (branch: string) => KEY.exec(branch)?.[0]

/** Whether a text is a key and nothing else. */
export const isKey = (text: string) => new RegExp(`^${KEY.source}$`).test(text.trim())

/** The pin wins; with none, the branch's key; with neither, no focus. */
export const resolveFocus = (pin: Pin | undefined, branch: string | undefined): Focus | undefined => {
  if (pin?.kind === 'key') return { kind: 'key', key: pin.key, from: 'pin' }
  if (pin?.kind === 'text') return { kind: 'text', text: pin.text }
  const key = branch === undefined ? undefined : keyIn(branch)

  return key === undefined ? undefined : { kind: 'key', key, from: 'branch' }
}

/** Cuts a text to `max` characters (an emoji counts as one), ending in `…`. */
export const cut = (text: string, max: number) => {
  const chars = [...text]
  return chars.length <= max ? text : `${chars.slice(0, max - 1).join('')}…`
}

/** The status line: the item when found, `?` when not, the key alone while loading. */
export const statusText = (focus: Focus, item: Item | 'unknown' | undefined) => {
  if (focus.kind === 'text') return cut(`🎯 ${focus.text}`, MAX)
  if (item === 'unknown') return `🎫 ${focus.key} · ?`
  if (item === undefined) return `🎫 ${focus.key}`

  return cut(`🎫 ${item.key} · ${item.status} · ${item.summary}`, MAX)
}
```

// The trackers behind one interface: Jira through the Atlassian CLI, or
// through the Atlassian Rovo connector of claude.ai.

import type { EngineInterface } from 'claude-code'

import type { Item } from '../types'

export type Source = {
  name: 'acli' | 'rovo'
  /** The item, or undefined when this tracker does not have it or cannot be reached. */
  fetch(key: string): Promise<Item | undefined>
  /** Posts a comment; rejects when the tracker refuses it. */
  comment(key: string, body: string): Promise<void>
}

const parse = (text: string): unknown => {
  try {
    return JSON.parse(text)
  } catch {
    // Rovo may wrap the JSON in prose: take the outermost object or array.
    const start = text.search(/[[{]/)
    const end = Math.max(text.lastIndexOf('}'), text.lastIndexOf(']'))
    if (start < 0 || end <= start) return undefined
    try {
      return JSON.parse(text.slice(start, end + 1))
    } catch {
      return undefined
    }
  }
}

/** A Jira issue JSON (`{ key, fields: { summary, status: { name } } }`) as an Item. */
const itemOf = (value: unknown): Item | undefined => {
  const issue = value as { key?: unknown; fields?: { summary?: unknown; status?: { name?: unknown } } } | undefined
  const key = issue?.key
  const summary = issue?.fields?.summary
  const status = issue?.fields?.status?.name

  return typeof key === 'string' && typeof summary === 'string' && typeof status === 'string' ? { key, summary, status } : undefined
}

export const parseAcliItem = (stdout: string) => itemOf(parse(stdout))
export const parseRovoItem = (text: string) => itemOf(parse(text))

export const parseRovoSites = (text: string): { cloudId: string; url: string }[] => {
  const sites = parse(text)
  if (!Array.isArray(sites)) return []

  return sites.flatMap(site =>
    typeof site?.id === 'string' && typeof site?.url === 'string' ? [{ cloudId: site.id as string, url: site.url as string }] : [],
  )
}

/** The text of a tool result: a string, MCP content blocks, or anything else as JSON. */
export const textOf = (result: unknown): string => {
  if (typeof result === 'string') return result
  const blocks = Array.isArray(result) ? result : (result as { content?: unknown } | undefined)?.content
  if (Array.isArray(blocks)) {
    return blocks
      .flatMap(block => (typeof block?.text === 'string' ? [block.text as string] : []))
      .join('\n')
  }

  return JSON.stringify(result)
}

export const acliSource = ($: EngineInterface): Source => ({
  name: 'acli',
  async fetch(key) {
    const ran = await $.process.run(['acli', 'jira', 'workitem', 'view', key, '--json']).catch(() => undefined)
    return ran?.exitCode === 0 ? parseAcliItem(ran.stdout) : undefined
  },
  async comment(key, body) {
    const ran = await $.process.run(['acli', 'jira', 'workitem', 'comment', 'create', '--key', key, '--body', body])
    if (ran.exitCode !== 0) throw new Error(ran.stderr.trim() || `acli exited ${ran.exitCode}`)
  },
})

/** The Rovo tool by its short name, whichever Rovo connector is authenticated. */
async function rovoTool($: EngineInterface, name: string) {
  const tools = await $.tool.list().catch(() => [])
  return tools.find(tool => /Atlassian_Rovo/i.test(tool.name) && tool.name.endsWith(`__${name}`))?.name
}

export async function rovoSites($: EngineInterface) {
  const tool = await rovoTool($, 'getAccessibleAtlassianResources')
  if (tool === undefined) return []
  const called = await $.tool.call({ tool } as never).catch(() => undefined)

  return called?.result === undefined ? [] : parseRovoSites(textOf(called.result))
}

export async function rovoSource($: EngineInterface, cloudId: string): Promise<Source | undefined> {
  const [get, add] = await Promise.all([rovoTool($, 'getJiraIssue'), rovoTool($, 'addCommentToJiraIssue')])
  if (get === undefined || add === undefined) return undefined

  return {
    name: 'rovo',
    async fetch(key) {
      const called = await $.tool.call({ tool: get, cloudId, issueIdOrKey: key } as never).catch(() => undefined)
      return called?.result === undefined ? undefined : parseRovoItem(textOf(called.result))
    },
    async comment(key, body) {
      const called = await $.tool.call({ tool: add, cloudId, issueIdOrKey: key, commentBody: body } as never)
      if (called.deny !== undefined) throw new Error(called.deny)
    },
  }
}
```

async function projectOf($: EngineInterface) {

async function saveConfig($: EngineInterface, root: string, config: ProjectConfig) {

/**
 * The project's source for a key: the stored one, else the first of acli and
 * Rovo (each site in turn) that has the key, stored for next time.
 */
async function sourceFor($: EngineInterface, root: string, key: string): Promise<{ source?: Source; item?: Item }> {
  const config = await configOf($, root)
  if (config.source === 'manual') return {}
  if (config.source === 'acli') return { source: acliSource($) }
  if (config.source === 'rovo' && config.cloudId !== undefined) return { source: await rovoSource($, config.cloudId) }

  const acli = acliSource($)
  const viaAcli = config.source === undefined ? await acli.fetch(key) : undefined
  if (viaAcli !== undefined) {
    await saveConfig($, root, { ...config, source: 'acli' })
    return { source: acli, item: viaAcli }
  }
  for (const site of await rovoSites($)) {
    const rovo = await rovoSource($, site.cloudId)
    const item = await rovo?.fetch(key)
    if (rovo !== undefined && item !== undefined) {
      await saveConfig($, root, { ...config, source: 'rovo', cloudId: site.cloudId })
      return { source: rovo, item }
    }
  }

  return {}
}

async function itemFor($: EngineInterface, root: string, key: string): Promise<Item | 'unknown'> {
  const cached = cache.get(key)
  if (cached !== undefined && Date.now() - cached.at < CACHE_MS) return cached.item
  const { source, item } = await sourceFor($, root, key)
  const found = item ?? (await source?.fetch(key)) ?? 'unknown'
  cache.set(key, { at: Date.now(), item: found })

  return found
}

function show($: EngineInterface, text: string | undefined) {
  if (text === shown) return
  shown = text
  $.ui.status(text)
}

/** Reads the focus, draws it at once, then with its item when one is fetched. */
async function refresh($: EngineInterface): Promise<Focus | undefined> {
  const mine = ++generation
  const root = await projectOf($)
  const [config, branch] = await Promise.all([configOf($, root), branchOf($, root)])
  const focus = resolveFocus(config.pin, branch)
  if (mine !== generation) return focus
  await update($, focusAtom, () => focus ?? null)
  if (focus === undefined) {
    show($, undefined)
    return focus
  }
  if (focus.kind === 'text') {
    show($, statusText(focus, undefined))
    return focus
  }
  const cached = cache.get(focus.key)
  show($, statusText(focus, cached?.item))
  const item = await itemFor($, root, focus.key)
  if (mine === generation) show($, statusText(focus, item))

  return focus
}

    return started
